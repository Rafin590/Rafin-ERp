Google Sheet কানেক্ট করার নিয়ম

সহজে ১ মিনিটে আপনার গুগল শিটের সাথে Rafin ERP কানেক্ট করুন

১. আপনার গুগল শিটে যান:

আপনার একটি Google Sheet ওপেন করে উপরে মেনু থেকে Extensions → Apps Script-এ ক্লিক করুন।

২. নিচের কোডটি কপি করে Apps Script-এ পেস্ট করুন:

function doPost(e) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = ss.getSheetByName("Orders") || ss.getActiveSheet();
    
    // Auto-create header if empty
    if (sheet.getLastRow() === 0) {
      sheet.appendRow(["Invoice", "Date", "Customer Name", "Phone", "Address", "Items", "Delivery Charge", "Discount", "Advance", "Grand Total", "Courier", "Tracking", "Status", "Note", "Timestamp"]);
      sheet.getRange(1, 1, 1, 15).setFontWeight("bold").setBackground("#4f46e5").setFontColor("#ffffff");
    }
    
    var data = JSON.parse(e.postData.contents);
    
    if (data.action === "bulk_sync" && Array.isArray(data.orders)) {
      data.orders.forEach(function(o) {
        appendOrUpdateOrder(sheet, o);
      });
      return ContentService.createTextOutput(JSON.stringify({ status: "success", count: data.orders.length })).setMimeType(ContentService.MimeType.JSON);
    }
    
    appendOrUpdateOrder(sheet, data);
    return ContentService.createTextOutput(JSON.stringify({ status: "success", invoice: data.invoice })).setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ status: "error", message: err.toString() })).setMimeType(ContentService.MimeType.JSON);
  }
}

function appendOrUpdateOrder(sheet, o) {
  var itemsStr = Array.isArray(o.items) ? o.items.map(function(i){ return i.name + " (x" + i.qty + ")"; }).join(", ") : (o.items || "");
  var row = [
    o.invoice || "",
    o.date || "",
    o.customerName || "",
    o.phone || "",
    o.address || "",
    itemsStr,
    o.deliveryCharge || 0,
    o.discount || 0,
    o.advance || 0,
    o.grandTotal || 0,
    o.courier || "",
    o.tracking || "",
    o.status || "Pending",
    o.note || "",
    new Date().toISOString()
  ];
  sheet.appendRow(row);
}

function doGet(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName("Orders") || ss.getActiveSheet();
  var data = sheet.getDataRange().getValues();
  if (data.length <= 1) {
    return ContentService.createTextOutput(JSON.stringify([])).setMimeType(ContentService.MimeType.JSON);
  }
  var headers = data[0];
  var results = [];
  for (var i = 1; i < data.length; i++) {
    var row = data[i];
    var obj = {};
    for (var j = 0; j < headers.length; j++) {
      obj[headers[j]] = row[j];
    }
    results.push(obj);
  }
  return ContentService.createTextOutput(JSON.stringify(results)).setMimeType(ContentService.MimeType.JSON);
}

৩. Deploy করুন:
